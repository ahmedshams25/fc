document.addEventListener('DOMContentLoaded', () => { // --- Elements Selection --- const addCompanyForm = document.getElementById('add-company-form'); const companyNameInput = document.getElementById('company-name'); const companiesList = document.getElementById('companies-list'); const addProjectForm = document.getElementById('add-project-form'); const projectCompanyIdInput = document.getElementById('project-company-id'); const projectModalEl = document.getElementById('addProjectModal'); const projectModal = new bootstrap.Modal(projectModalEl); // --- Data Management --- // Load companies from localStorage or initialize an empty array let companies = JSON.parse(localStorage.getItem('companies')) || []; // Function to save the current state of companies to localStorage const saveCompanies = () => { localStorage.setItem('companies', JSON.stringify(companies)); }; // --- Rendering Functions --- const renderProjects = (companyIndex) => { const company = companies[companyIndex]; if (!company || !company.projects) return ''; // Return empty string if no projects if (company.projects.length === 0) { return '

لا توجد مشاريع مضافة لهذه الشركة بعد.

'; } return company.projects.map((project, projectIndex) => `
${project.name}

التكلفة: ${project.cost || 'غير محدد'}

الموعد النهائي: ${project.deadline}

🔗 زيارة الرابط
`).join(''); }; const renderCompanies = () => { companiesList.innerHTML = ''; // Clear the list before rendering if (companies.length === 0) { companiesList.innerHTML = '
لم تقم بإضافة أي شركات بعد. ابدأ بإضافة شركة جديدة من النموذج.
'; return; } companiesList.innerHTML = companies.map((company, index) => `

${renderProjects(index)}
`).join(''); }; // --- Event Handlers --- // Handle adding a new company addCompanyForm.addEventListener('submit', (e) => { e.preventDefault(); const companyName = companyNameInput.value.trim(); if (companyName) { companies.push({ name: companyName, projects: [] }); saveCompanies(); renderCompanies(); addCompanyForm.reset(); } }); // Handle adding a new project addProjectForm.addEventListener('submit', (e) => { e.preventDefault(); const companyIndex = projectCompanyIdInput.value; if (companyIndex === '') return; const newProject = { name: document.getElementById('project-name').value, deadline: document.getElementById('project-deadline').value, cost: document.getElementById('project-cost').value, link: document.getElementById('project-link').value }; companies[companyIndex].projects.push(newProject); saveCompanies(); renderCompanies(); // Re-render everything to update the correct project list projectModal.hide(); }); // Using Event Delegation for dynamically created buttons companiesList.addEventListener('click', (e) => { const target = e.target; const companyIndex = target.dataset.companyIndex; // Click on "Add Project" button if (target.classList.contains('btn-add-project')) { projectCompanyIdInput.value = companyIndex; addProjectForm.reset(); projectModal.show(); } // Click on "Delete Company" button if (target.classList.contains('btn-delete-company')) { if (confirm('هل أنت متأكد من رغبتك في حذف هذه الشركة وكل مشاريعها؟')) { companies.splice(companyIndex, 1); saveCompanies(); renderCompanies(); } } // Click on "Delete Project" button if (target.classList.contains('btn-delete-project')) { const projectIndex = target.dataset.projectIndex; if (confirm('هل أنت متأكد من رغبتك في حذف هذا المشروع؟')) { companies[companyIndex].projects.splice(projectIndex, 1); saveCompanies(); renderCompanies(); } } }); // --- Initial Render --- renderCompanies(); });
