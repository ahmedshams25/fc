`; companiesList.appendChild(companyElement); renderProjects(index); }); }; const renderProjects = (companyIndex) => { const projectsListContainer = document.getElementById(`projects-list-${companyIndex}`); const company = companies[companyIndex]; projectsListContainer.innerHTML = ''; if (company.projects && company.projects.length > 0) { company.projects.forEach((project, projectIndex) => { const projectElement = document.createElement('div'); projectElement.classList.add('card', 'mb-2', 'project-card'); projectElement.innerHTML = `
